# Email Client — environment fallback configuration
#
# The preferred way to configure the app is `npm run setup`, which
# creates app.config.json (gitignored). These environment variables are
# only used as a fallback — when app.config.json is absent or a field
# is empty.

# Resend API key — https://resend.com/api-keys
RESEND_API_KEY=re_your_api_key_here

# Sender address — must be on a domain verified in Resend.
# Accepts either "mailer@example.com" or "Display Name <mailer@example.com>".
FROM_EMAIL=mailer@example.com
# FROM_NAME=Sender

# Optional: receives a silent copy of every email sent through the app
ADMIN_EMAIL=

# Optional: rate limiting
RATE_LIMIT_MAX=10
RATE_LIMIT_WINDOW_MS=60000

# Optional: JSON map of {{var}} placeholders used in email templates
# TEMPLATE_VARS={"senderName":"Sender","supportContact":"support"}
